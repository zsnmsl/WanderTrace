cmake_minimum_required(VERSION 3.18)
project(wander_step_pipeline_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(wander_step_pipeline_checks step_pipeline_checks.cpp)
if(MSVC)
    target_compile_options(wander_step_pipeline_checks PRIVATE /UNDEBUG)
else()
    target_compile_options(wander_step_pipeline_checks PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME step_pipeline COMMAND wander_step_pipeline_checks)
