cmake_minimum_required(VERSION 3.18)
project(wander_native LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(DOBBY_DEBUG "Dobby debug" OFF)
option(DOBBY_GENERATE_SHARED "Dobby shared" OFF)
option(Plugin.SymbolResolver "Dobby symbols" OFF)
option(Plugin.ImportTableReplace "Dobby import table" OFF)
add_subdirectory(vendor/Dobby)
add_library(wander_native SHARED wander_native.cpp)
target_include_directories(wander_native PRIVATE vendor/Dobby/include)
target_link_libraries(wander_native PRIVATE dobby log dl)
target_link_options(wander_native PRIVATE "-Wl,-z,max-page-size=16384")
option(WANDERTRACE_BUILD_CHECKS "Build standalone native checks" OFF)
if(WANDERTRACE_BUILD_CHECKS)
    add_executable(wander_native_checks tests/native_checks.cpp)
    target_include_directories(wander_native_checks PRIVATE vendor/Dobby/include)
    target_link_libraries(wander_native_checks PRIVATE dobby log dl)
    target_compile_options(wander_native_checks PRIVATE -UNDEBUG)
    add_executable(wander_device_abi_check tests/device_abi_check.cpp)
    target_include_directories(wander_device_abi_check PRIVATE vendor/Dobby/include)
    target_link_libraries(wander_device_abi_check PRIVATE dobby log dl)
endif()
